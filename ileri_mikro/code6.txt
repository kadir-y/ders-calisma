// HWREG(0x40025000 + maske 00) = deger
// 0x40025000 + 0000 0010 00 = 0x40025008
// 0x40025000 + 0000 1110 00 = 0x40025038
// 0x40025000 + 1111 1111 00 = 0x400253FC
// 0x40025000 + 0000 1110 00   = 0x40025038   (PF1, PF2, PF3) => LEDLER İÇİN BUNLAR YETERLİ


int ikininustal(int value){
    int i=1;
    int carpim=1;
    for (i=0;i<value;i++)
    {
        carpim=carpim*2;
    }

    return carpim;
}

void portfpin1set(int setdegeri)
{
    //GPIOPinWrite(GPIO_PORTF_BASE, GPIO_PIN_1, GPIO_PIN_1);
    HWREG(0x400253fc)=setdegeri;

}

void portfpinXset(int pinno,int setdegeri)
{
    GPIOPinWrite(GPIO_PORTF_BASE, pinno, setdegeri);
    //HWREG()=setdegeri;
}

void portset(long portvalue, int pinno,int setdegeri)
{
    //GPIOPinWrite(portvalue, pinno, setdegeri);
    HWREG(portvalue+4*ikininustal(pinno))=0;
}
