//Offsetler + Base adres = Ram Adres'i

void GpioInitSettings(){

    SYSCTL_RCGCGPIO_R = SYSCTL_RCGCGPIO_R | 0x20; // Portf Aktif
   // SYSCTL_RCGCGPIO_R = SYSCTL_RCGCGPIO_R (1<<5);
    GPIO_PORTF_LOCK_R = 0x4C4F434B;
    GPIO_PORTF_CR_R = 0x01;
    GPIO_PORTF_DIR_R |= 0x0E; // pin0 ile pin4 input , 1-2-3 output
   // GPIO_PORTF_DIR_R |= (1<<1) |(1<<2) | (1<<3);
    GPIO_PORTF_AFSEL_R = 0x00;
    GPIO_PORTF_DEN_R = 0x1F;

    GPIO_PORTF_PUR_R |= 0x11;
}


int main(void)
{
    GpioInitSettings();

    while(1){

      if ((GPIO_PORTF_DATA_R & 0x01) == 0){
          GPIO_PORTF_DATA_R |= ~0b1000;
      }

      else if ((GPIO_PORTF_DATA_R & 0x10) == 0){
          GPIO_PORTF_DATA_R &= ~0b0111;

      }
    }
}
